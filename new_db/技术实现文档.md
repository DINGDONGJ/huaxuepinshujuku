```sql
    化学品ID INT NOT NULL,
    章节序号 INT NOT NULL,
    章节标题 VARCHAR(200) NOT NULL,
```

```python
def download_image(img_url, output_dir, base_url):
    """
    下载图片并保存到本地
    
    参数:
        img_url: 图片URL（可能是相对路径）
        output_dir: 输出目录
        base_url: 基础URL，用于解析相对路径
        
    返回:
        相对路径字符串，如 'images/abc123.png'
    """
    # 处理相对路径
    if not img_url.startswith('http'):
        img_url = urljoin(base_url, img_url)
    
    # 使用URL哈希作为文件名，避免重复和特殊字符问题
    url_hash = hashlib.md5(img_url.encode()).hexdigest()[:16]
    ext = os.path.splitext(urlparse(img_url).path)[1] or '.png'
    filename = f"{url_hash}{ext}"
    
    # 保存到 images 子目录
    images_dir = os.path.join(output_dir, 'images')
    os.makedirs(images_dir, exist_ok=True)
    
    filepath = os.path.join(images_dir, filename)
    relative_path = f'images/{filename}'
    
    # 下载并保存
    response = requests.get(img_url, timeout=10)
    if response.status_code == 200:
        with open(filepath, 'wb') as f:
            f.write(response.content)
        return relative_path
    return None
```

```python
def scrape_msds_part(mid, part_num, output_dir):
    """
    爬取指定章节的MSDS内容和图片
    
    返回:
        (text_content, images)
        - text_content: 文本内容
        - images: 图片列表 [{"url": "...", "alt": "...", "type": "..."}]
    """
    # ... HTML解析代码 ...
    
    # 提取图片
    temp_images = []
    for img in soup.find_all('img'):
        img_url = img.get('src', '')
        img_alt = img.get('alt', '')
        
        # 过滤掉不需要的图片
        if any(skip in img_url.lower() for skip in ['logo', 'icon', 'avatar']):
            continue
        if any(skip in img_alt.lower() for skip in ['logo', 'icon']):
            continue
            
        # 下载图片
        relative_path = download_image(img_url, output_dir, base_url)
        if relative_path:
            # 判断图片类型
            img_type = 'ghs' if 'ghs' in img_alt.lower() or part_num == 2 else \
                      'transport' if part_num == 14 else 'other'
            
            temp_images.append({
                'url': relative_path,
                'alt': img_alt or f'图片{len(temp_images)+1}',
                'type': img_type,
                'original_url': img_url
            })
    
    # ⚠️ 重要：过滤掉最后一张图片（通常是二维码）
    images = temp_images[:-1] if len(temp_images) > 1 else []
    
    return text_content, images
```

**关键过滤规则：**
- 排除logo、icon等装饰性图片
- 排除最后一张图片（二维码）

```javascript
function insertImagesIntoText(textContent, imagesHtml, chapterNum) {
```

```javascript
function formatFirstChapter(content) {
    /**
     * 格式化第一章节：化学品及企业标识
     * 特点：带标签的字段，需要加粗标签
     */
    let html = '';
    const lines = content.split('\n');
    
    // 定义固定字段
    const fixedFields = [
        '中文名', '英文名', '分子式', '相对分子质量',
        'CAS号', '危险性类别', '化学品俗名或商品名',
        '供应商', '应急咨询电话', '传真', '邮箱', '地址'
    ];
    
    for (let i = 0; i < lines.length; i++) {
        const line = lines[i].trim();
        if (!line) continue;
        
        // 匹配 "标签: 内容" 或 "标签：内容"
        const fieldMatch = line.match(/^([^:：]+)[：:]\s*(.*)$/);
        
        if (fieldMatch) {
            const label = fieldMatch[1].trim();
            const value = fieldMatch[2].trim();
            
            if (fixedFields.includes(label)) {
                // 固定字段：蓝色加粗标签
                html += `
                    <div class="msds-field">
                        <div class="msds-field-label">${label}</div>
                        <div class="msds-field-value">${value || '无'}</div>
                    </div>
                `;
            } else {
                // 其他字段：普通显示
                html += `<div style="margin-bottom: 8px;">${line}</div>`;
            }
        } else {
            // 非字段行：作为标题或普通文本
            html += `<div class="msds-section-title">${line}</div>`;
        }
    }
    
    return html;
}
```

#### 5. 第二章节格式化（重点）

```javascript
function formatSecondChapter(content) {
    /**
     * 格式化第二章节：危险性概述
     * 
     * 特殊格式处理：
     * 1. GHS标签要素 - 蓝色加粗标题
     * 2. 毒性类别、健康危害 - 蓝色加粗小标题
     * 3. H编号（H301等）- 蓝色加粗文字，说明普通文字
     * 4. P编号（P301+P310等）- 蓝色加粗文字，说明普通文字
     * 5. 数字列表 - 普通显示
     * 
     * ⚠️ 关键难点：空行处理
     * - 原始数据中H/P编号和说明可能在不同行
     * - HTML生成必须单行，不能有换行符和缩进
     * - 否则会产生大量空白行
     */
    let html = '';
    const lines = content.split('\n');
    
    // GHS子标题列表
    const ghsSubtitles = [
        '信号词', '危险性类别', '象形图', '危险性说明', 
        '防范说明', '物理和化学危险', '健康危害', '环境危害'
    ];
    
    // 健康类别列表
    const healthCategories = [
        '急性毒性-经口', '急性毒性-经皮', '急性毒性-吸入',
        '皮肤腐蚀/刺激', '严重眼损伤/眼刺激', '呼吸道致敏',
        '皮肤致敏', '生殖细胞致突变性', '致癌性',
        '生殖毒性', '特异性靶器官毒性-一次接触',
        '特异性靶器官毒性-反复接触', '吸入危害'
    ];
    
    for (let i = 0; i < lines.length; i++) {
        const line = lines[i].trim();
        if (!line) continue;
        
        // 1. 主标题（GHS标签要素）
        if (line.includes('GHS') && line.includes('标签')) {
            html += `<div class="msds-section-title" style="margin-top: 0;">${line}</div>`;
            continue;
        }
        
        // 2. GHS子标题
        if (ghsSubtitles.includes(line)) {
            html += `<div class="msds-section-title" style="font-size: 14px;">${line}</div>`;
            continue;
        }
        
        // 3. 健康类别（可能带"类别 X"）
        const matchedCategory = healthCategories.find(cat => line.includes(cat));
        if (matchedCategory) {
            if (line === matchedCategory) {
                // 只有类别名：作为小标题
                html += `<div class="msds-section-title" style="margin-top: 12px; font-size: 14px; color: #667eea;">${line}</div>`;
                continue;
            } else if (line.includes('类别')) {
                // 包含"类别 X"：分开显示
                const parts = line.split(/\s+类别\s+/);
                if (parts.length === 2) {
                    // ⚠️ 注意：单行HTML，无换行符
                    html += `<div style="margin-bottom: 8px; margin-top: 8px;"><div class="msds-section-title" style="font-size: 14px; color: #667eea; margin: 0; line-height: 1.5;">${parts[0]}</div><div style="color: #495057; margin-top: 4px; padding-left: 15px; line-height: 1.5;">类别 ${parts[1]}</div></div>`;
                    continue;
                }
            }
        }
        
        // 4. H编号处理（支持多行）
        if (/^H\d+/.test(line)) {
            const match = line.match(/^(H\d+)\s*(.*)$/);
            if (match) {
                let code = match[1];
                let description = match[2];
                
                // 如果当前行没有说明，尝试读取下一行
                if (!description && i + 1 < lines.length) {
                    const nextLine = lines[i + 1].trim();
                    // 确保下一行不是另一个编号
                    if (nextLine && !/^[HP]\d+/.test(nextLine) && !/^\d+\./.test(nextLine)) {
                        description = nextLine;
                        i++; // 跳过下一行
                    }
                }
                
                // 只有有说明才输出（避免空行）
                if (description) {
                    // ⚠️ 关键：单行HTML，line-height: 1.5 控制间距
                    html += `<div style="margin-bottom: 8px; word-wrap: break-word; overflow-wrap: break-word; line-height: 1.5;"><span style="color: #667eea; font-weight: 600; margin-right: 8px;">${code}</span><span style="color: #495057; word-wrap: break-word; overflow-wrap: break-word;">${description}</span></div>`;
                }
                continue;
            }
        }
        
        // 5. P编号处理（支持 P301+P310 格式，支持多行）
        if (/^P[\d+]+/.test(line)) {
            const match = line.match(/^(P[\d+]+)\s*(.*)$/);
            if (match) {
                let code = match[1];
                let description = match[2];
                
                // 处理多行P编号
                let nextIndex = i + 1;
                while (nextIndex < lines.length) {
                    const nextLine = lines[nextIndex].trim();
                    
                    // 空行：跳过
                    if (!nextLine) {
                        nextIndex++;
                        continue;
                    }
                    
                    // 纯数字或P+数字：组合编号的一部分，跳过
                    if (/^P?\d+\+?$/.test(nextLine)) {
                        nextIndex++;
                        continue;
                    }
                    
                    // 另一个编号：停止
                    if (/^[HP]\d+/.test(nextLine) || /^\d+\./.test(nextLine)) {
                        break;
                    }
                    
                    // 说明文字：保存
                    if (!description) {
                        description = nextLine;
                        nextIndex++;
                    }
                    break;
                }
                
                // 更新循环索引
                i = nextIndex - 1;
                
                // 只有有说明才输出（避免空行）
                if (description) {
                    // ⚠️ 关键：单行HTML
                    html += `<div style="margin-bottom: 8px; word-wrap: break-word; overflow-wrap: break-word; line-height: 1.5;"><span style="color: #667eea; font-weight: 600; margin-right: 8px;">${code}</span><span style="color: #495057; word-wrap: break-word; overflow-wrap: break-word;">${description}</span></div>`;
                }
                continue;
            }
        }
        
        // 6. 数字列表（1. 2. 3.）
        if (/^\d+\.\s/.test(line)) {
            html += `<div style="margin-bottom: 8px; padding-left: 10px;">${line}</div>`;
            continue;
        }
        
        // 7. 过滤纯数字行（P编号组合的残留）
        if (/^P?\d+\+?$/.test(line)) {
            continue;
        }
        
        // 8. 其他普通文本
        html += `<div style="margin-bottom: 8px;">${line}</div>`;
    }
    
    return html;
}
```

## 🐛 问题解决记录

### 问题1：空行过多

**现象：**
- MSDS章节显示时有大量空白行
- 特别是在消防措施、危险性概述等章节

**原因：**
- CSS使用 `white-space: pre-wrap` 保留了所有换行符
- 原始数据中存在多余的空行

**解决方案：**
```css
/* 改为 pre-line，只保留实际换行，忽略多余空格 */
.chapter-text {
    white-space: pre-line;
    word-wrap: break-word;
}
```

### 问题2：点击内容区域会折叠

**现象：**
- 展开章节后，点击内容任意位置都会折叠

**原因：**
- `onclick` 事件绑定在父容器 `.chapter-item` 上
- 点击事件冒泡导致内容区域的点击也触发折叠

**解决方案：**
```html
<!-- 只在header上绑定点击 -->
<div class="chapter-item">
    <div class="chapter-header" onclick="toggleChapter(this)">
        ...
    </div>
    <div class="chapter-content">
        <!-- 这里点击不会触发 -->
    </div>
</div>
```

### 问题5：H/P编号不显示蓝色

**现象：**

3. CSS样式被覆盖

**解决方案：**

```javascript
    if (nextLine && !/^[HP]\d+/.test(nextLine)) {
        description = nextLine;
        i++; // 跳过下一行
    }
}

// 3. 使用行内样式，优先级更高
html += `<span style="color: #667eea; font-weight: 600;">${code}</span>`;
```

### 问题6：第二章节显示不全

**现象：**
- 内容超出容器，末尾被截断
- 添加蓝色样式后问题加重

**原因：**
- `.chapter-item.active .chapter-content` 的 `max-height: 5000px` 不够
- 长文本没有自动换行

### 问题7：HTML模板中的空行（最终问题）

**现象：**
- 修复了所有逻辑后，H/P编号之间仍有很大空白
- line-height已设置为1.5，但实际高度达到67.5px

**根本原因：**
```javascript
// ❌ 错误：HTML模板字符串包含换行和缩进
html += `
    <div style="line-height: 1.5;">
        <span>H301</span>
        <span>说明</span>
    </div>
`;

// 这段HTML被浏览器解析为3行内容：
// 第1行：换行+空格
// 第2行：<span>H301</span>换行+空格  
// 第3行：<span>说明</span>换行+空格
// 总高度 = 1.5 * 3 = 4.5倍 = 67.5px（字体15px）
```

**最终解决方案：**
```javascript
// ✅ 正确：单行HTML，无换行无缩进
html += `<div style="line-height: 1.5;"><span>H301</span><span>说明</span></div>`;

// 浏览器解析为1行内容
// 总高度 = 1.5 * 1 = 22.5px
```

**教训：**
- JavaScript模板字符串中的换行符 `\n` 会被保留
- 在HTML中，换行符会被转换为空格，但仍占据行高
- 生成HTML时要么完全单行，要么使用 `.trim()` 去除首尾空白

## 🎯 全局功能：CAS号高亮

### 需求背景

化学品的CAS号（CAS Registry Number）是重要的标识符，需要在整个Web应用中以**红色加粗**显示，便于用户快速识别。

### 实现方案

#### 1. CSS样式定义

**文件位置**: `templates/index.html` → `<style>`

```css
.cas-number {
    color: #dc3545;  /* 红色 */
    font-weight: 700; /* 加粗 */
}
```

#### 2. JavaScript高亮函数

**文件位置**: `templates/index.html` → `highlightCASNumbers()`

```javascript
function highlightCASNumbers(html) {
    if (!html) return html;
    
    // CAS号正则表达式：XX-XX-X 或 XXXX-XX-X 格式
    const casRegex = /\b(\d{2,7}-\d{2}-\d)\b/g;
    
    // 替换为带样式的span标签
    return html.replace(casRegex, '<span class="cas-number">$1</span>');
}
```

**正则表达式说明**：
- `\b`: 单词边界，确保完整匹配
- `\d{2,7}`: 2-7位数字
- `-\d{2}`: 连字符 + 2位数字
- `-\d`: 连字符 + 1位数字（校验位）
- `\b`: 单词边界
- `g`: 全局匹配

**支持的CAS号格式**：
- `50-00-0` ✅
- `64-17-5` ✅
- `7732-18-5` ✅
- `123456-78-9` ✅

#### 3. 应用位置

**位置1：基本信息卡片**

```javascript
// 在 displayResults 函数中
<div class="info-item">
    <div class="info-label">CAS号</div>
    <div class="info-value">${highlightCASNumbers(basic.CAS号)}</div>
</div>
```

**位置2：所有章节内容**

```javascript

    // ... 格式化逻辑 ...
    
    // 最后应用CAS号高亮

    }
    
    // 对未格式化的内容也应用高亮
    return highlightCASNumbers(text);
}
```
