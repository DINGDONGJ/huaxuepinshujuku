-- 完整统计
SELECT 
    '修改后统计' AS 说明,
    COUNT(DISTINCT c.编号) AS 化学品总数,
    COUNT(*) AS 第1章记录总数,
    SUM(CASE WHEN s.内容 LIKE '%新企业名称%' THEN 1 ELSE 0 END) AS 包含新企业信息数,
    SUM(CASE WHEN s.内容 LIKE '%常州合规思远%' THEN 1 ELSE 0 END) AS 仍包含旧企业信息数
FROM 化学品 c
LEFT JOIN MSDS文档 m ON c.编号 = m.化学品编号
LEFT JOIN MSDS章节 s ON m.编号 = s.文档编号 AND s.章节序号 = 1;

-- 查看所有化学品的更新情况
SELECT 
    c.中文名,
    c.CAS号,
    CASE 
        WHEN s.内容 LIKE '%新企业名称%' THEN '已更新'
        WHEN s.内容 LIKE '%常州合规思远%' THEN '未更新'
        ELSE '无第1章数据'
    END AS 更新状态
FROM 化学品 c
LEFT JOIN MSDS文档 m ON c.编号 = m.化学品编号
LEFT JOIN MSDS章节 s ON m.编号 = s.文档编号 AND s.章节序号 = 1
ORDER BY c.中文名;

================================================================================
附录D：常见问题解答
================================================================================

Q1: 修改后Web界面看不到变化？
A1: 刷新浏览器页面（Ctrl+F5强制刷新），或者清除浏览器缓存。

Q2: 如何确认修改已生效？
A2: 在Web界面搜索任意化学品，查看第1章"化学品及企业标识"是否显示新的企业信息。

Q3: 修改会影响导出的JSON文件吗？
A3: 不会。msds_json文件夹中的JSON文件是独立的，需要单独修改或重新爬取。

Q4: 电话和传真号码相同怎么办？
A4: 分两步操作：
    - 第一步：REPLACE('0519-85150306', '新电话::临时标记::')
    - 第二步：根据上下文分别替换电话和传真部分

Q5: 如何只修改编制单位，不修改企业名称？
A5: 只执行MSDS文档表的UPDATE：
    UPDATE MSDS文档
    SET 编制单位 = '新编制单位'
    WHERE 编制单位 = '合规化学网';

Q6: 备份文件太大怎么办？
A6: 使用压缩：
    mysqldump -u root -p1234 危化品简化数据库 | gzip > backup.sql.gz

Q7: 修改后能恢复到之前的状态吗？
A7: 可以。只要有备份文件，随时可以恢复：
    mysql -u root -p1234 危化品简化数据库 < backup.sql
